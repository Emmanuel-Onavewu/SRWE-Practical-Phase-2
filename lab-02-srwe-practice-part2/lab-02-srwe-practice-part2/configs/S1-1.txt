! Configuration commands used in this lab (not a running-config export)
vlan 10
 name users
vlan 999
 name unused

ip dhcp snooping
ip dhcp snooping vlan 10,999
ip arp inspection vlan 10,999

interface range FastEthernet0/1-5
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 4
 switchport port-security violation restrict
 switchport port-security aging time 10
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping limit rate 5

interface GigabitEthernet0/1
 description Link to Central G0/0/0
 switchport mode access
 switchport access vlan 10
 switchport port-security
 switchport port-security maximum 4
 switchport port-security violation restrict
 switchport port-security aging time 10
 switchport port-security mac-address sticky
 spanning-tree portfast
 spanning-tree bpduguard enable
 ip dhcp snooping trust
 ip dhcp snooping limit rate 5
 ip arp inspection trust

interface FastEthernet0/1
 switchport port-security mac-address <PC-A-MAC>

interface range FastEthernet0/6-24, GigabitEthernet0/2
 description Unused port - shutdown
 switchport mode access
